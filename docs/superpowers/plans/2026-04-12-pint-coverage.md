```js
// lib/siPrefixes.js
"use strict";

var SI_PREFIXES = {
  y:  { name: "Yocto",  factor: 1e-24, latex: "$y$" },
  z:  { name: "Zepto",  factor: 1e-21, latex: "$z$" },
  a:  { name: "Atto",   factor: 1e-18, latex: "$a$" },
  f:  { name: "Femto",  factor: 1e-15, latex: "$f$" },
  p:  { name: "Pico",   factor: 1e-12, latex: "$p$" },
  n:  { name: "Nano",   factor: 1e-9,  latex: "$n$" },
  μ:  { name: "Micro",  factor: 1e-6,  latex: "${\\mu}" },
  m:  { name: "Milli",  factor: 1e-3,  latex: "$m$" },
  c:  { name: "Centi",  factor: 1e-2,  latex: "$c$" },
  d:  { name: "Deci",   factor: 1e-1,  latex: "$d$" },
  da: { name: "Deca",   factor: 1e1,   latex: "$da$" },
  h:  { name: "Hecto",  factor: 1e2,   latex: "$h$" },
  k:  { name: "Kilo",   factor: 1e3,   latex: "$k$" },
  M:  { name: "Mega",   factor: 1e6,   latex: "$M$" },
  G:  { name: "Giga",   factor: 1e9,   latex: "$G$" },
  T:  { name: "Tera",   factor: 1e12,  latex: "$T$" },
  P:  { name: "Peta",   factor: 1e15,  latex: "$P$" },
  E:  { name: "Exa",    factor: 1e18,  latex: "$E$" }
};

function expandSI(baseAbbr, baseName, baseToAnchor, prefixes, options) {
  options = options || {};
  var pluralName = options.pluralName || (baseName + "s");
  var baseLatex = options.baseLatex || baseAbbr;

  var result = {};

  // Add the base unit itself
  result[baseAbbr] = {
    name: { singular: baseName, plural: pluralName },
    to_anchor: baseToAnchor,
    _latex: "$" + baseLatex + "$"
  };

  // Add each prefixed variant
  for (var i = 0; i < prefixes.length; i++) {
    var sym = prefixes[i];
    var pre = SI_PREFIXES[sym];
    if (!pre) {
      throw new Error("Unknown SI prefix: " + sym);
    }

    var abbr = sym + baseAbbr;
    var singular = pre.name.toLowerCase() + baseName.toLowerCase();
    // Capitalize first letter
    singular = singular.charAt(0).toUpperCase() + singular.slice(1);

    var plural;
    if (options.pluralName) {
      plural = pre.name.toLowerCase() + options.pluralName.toLowerCase();
      plural = plural.charAt(0).toUpperCase() + plural.slice(1);
    } else {
      plural = singular + "s";
    }

    // LaTeX: for micro, use ${\\mu}ABBR$; for others, $prefixABBR$
    var latex;
    if (sym === "μ") {
      latex = "${\\mu}" + baseLatex + "$";
    } else {
      latex = "$" + sym + baseLatex + "$";
    }

    result[abbr] = {
      name: { singular: singular, plural: plural },
      to_anchor: baseToAnchor * pre.factor,
      _latex: latex
    };
  }

  return result;
}

module.exports = { expandSI: expandSI, SI_PREFIXES: SI_PREFIXES };
```

```js
  getUnit(abbr) {
    if (!Converter._unitCache) {
      var cache = {};
      each(measures, function (systems, measure) {
        each(systems, function (units, system) {
          if (system === "_anchors") return false;
          each(units, function (unit, unitAbbr) {
            if (!cache[unitAbbr]) {
              cache[unitAbbr] = {
                abbr: unitAbbr,
                measure: measure,
                system: system,
                unit: unit,
```

```js
// scripts/check-collisions.js
"use strict";
var convert = require("../lib");

var seen = {};
var collisions = [];
var allUnits = convert().list();

allUnits.forEach(function (unit) {
  if (seen[unit.abbr]) {
    collisions.push({
      abbr: unit.abbr,
      measures: [seen[unit.abbr], unit.measure],
    });
  } else {
    seen[unit.abbr] = unit.measure;
  }
});

if (collisions.length > 0) {
  console.error("COLLISION DETECTED: " + collisions.length + " abbreviation(s) appear in multiple measures:");
  collisions.forEach(function (c) {
    console.error("  " + c.abbr + " -> " + c.measures.join(", "));
  });
  process.exit(1);
} else {
  console.log("OK: " + Object.keys(seen).length + " unit abbreviations, 0 collisions.");
  process.exit(0);
}
```

```js
// scripts/gen-registry.js
"use strict";
var fs = require("fs");
var path = require("path");
var convert = require("../lib");
var pkg = require("../package.json");

var allUnits = convert().list();
var allMeasures = convert().measures();

// Build units-registry.json
var registry = {
  version: pkg.version,
  generatedAt: new Date().toISOString(),
  measures: {},
};

allUnits.forEach(function (unit) {
  if (!registry.measures[unit.measure]) {
    registry.measures[unit.measure] = {
      systems: [],
      units: {},
      anchors: {},
    };
  }
  var m = registry.measures[unit.measure];
  if (m.systems.indexOf(unit.system) === -1) {
    m.systems.push(unit.system);
  }

  // Read _latex from the raw definition if available
  var rawDefs = require("../lib");
  // We need direct access to measures - load via index internals
  // Instead, just use abbr as latex fallback
  var latex = "$" + unit.abbr + "$";

  m.units[unit.abbr] = {
    system: unit.system,
    singular: unit.singular,
    plural: unit.plural,
    latex: latex,
  };
});

// Build latexUnits.json
var latexUnits = { latexUnits: { "--": "$--$" } };
allUnits.forEach(function (unit) {
  latexUnits.latexUnits[unit.abbr] = "$" + unit.abbr + "$";
});

// Build latexUnitTypes.json
var latexUnitTypes = {
  latexUnitTypes: [
    { id: 0, name: "-Select-", siunit: "--", imperialunit: "--" },
    { id: 1, name: "--", siunit: "--", imperialunit: "--" },
  ],
};

var nextId = 2;
allMeasures.forEach(function (measure) {
  var possibilities = convert().possibilities(measure);
  var siunit = possibilities[0] || "--";
  var imperialunit = "--";

  // Find first imperial unit if any
  var units = convert().list(measure);
  for (var i = 0; i < units.length; i++) {
    if (units[i].system === "metric" && siunit === possibilities[0]) {
      siunit = units[i].abbr;
      break;
    }
  }
  for (var j = 0; j < units.length; j++) {
    if (units[j].system !== "metric") {
      imperialunit = units[j].abbr;
      break;
    }
  }

  latexUnitTypes.latexUnitTypes.push({
    id: nextId++,
    name: measure,
    siunit: siunit,
    imperialunit: imperialunit,
  });
});

// Write files
var distDir = path.join(__dirname, "..", "dist");
if (!fs.existsSync(distDir)) {
  fs.mkdirSync(distDir, { recursive: true });
}

fs.writeFileSync(
  path.join(distDir, "units-registry.json"),
  JSON.stringify(registry, null, 2) + "\n"
);
fs.writeFileSync(
  path.join(distDir, "latexUnits.json"),
  JSON.stringify(latexUnits, null, 4) + "\n"
);
fs.writeFileSync(
  path.join(distDir, "latexUnitTypes.json"),
  JSON.stringify(latexUnitTypes, null, 4) + "\n"
);

console.log(
  "Generated: units-registry.json (" +
    Object.keys(registry.measures).length +
    " measures), latexUnits.json (" +
    Object.keys(latexUnits.latexUnits).length +
    " entries), latexUnitTypes.json (" +
    latexUnitTypes.latexUnitTypes.length +
    " entries)"
);
```

```python
"""Generate pint-reference.json from Python Pint for cross-validation."""
import json
import pint

ureg = pint.UnitRegistry()

# Map of convert-units abbreviations to Pint unit names
# Only include units that exist in both libraries
UNIT_MAP = {
    "Length": {
        "m": "meter", "km": "kilometer", "cm": "centimeter",
        "mm": "millimeter", "μm": "micrometer", "nm": "nanometer",
        "in": "inch", "ft": "foot", "yd": "yard", "mi": "mile",

        "atm": "atmosphere",
    },
    "Energy": {
        "J": "joule", "kJ": "kilojoule",
        "cal": "calorie", "kcal": "kilocalorie",
        "eV": "electron_volt",
    },
    "Force": {
        "N": "newton", "kN": "kilonewton", "dyn": "dyne",
        "lbf": "pound_force",
    },
    "Power": {
        "W": "watt", "kW": "kilowatt", "MW": "megawatt",
    },
    "Frequency": {
        "Hz": "hertz", "kHz": "kilohertz", "MHz": "megahertz",
        "GHz": "gigahertz",
    },
    "Time": {
        "s": "second", "min": "minute", "h": "hour", "d": "day",
    },
    "Angle": {
        "deg": "degree", "rad": "radian",
    },
}

reference = {}
for measure, units in UNIT_MAP.items():
    if not units:
        continue
    reference[measure] = {}

    # Find anchor (first unit in the dict)
    anchor_abbr = list(units.keys())[0]
    anchor_pint = units[anchor_abbr]

    for abbr, pint_name in units.items():
        try:
            qty = ureg.Quantity(1, pint_name)
            converted = qty.to(anchor_pint).magnitude
            reference[measure][abbr] = {
                "to_anchor_pint": converted,
                "anchor": anchor_abbr,
                "pint_unit": pint_name,
            }
        except Exception as e:
            reference[measure][abbr] = {"error": str(e)}

with open("scripts/pint-reference.json", "w") as f:
    json.dump(reference, f, indent=2)

print(f"Generated pint-reference.json with {sum(len(v) for v in reference.values())} unit entries")
```

- [ ] **Step 2: Create the JS verification script**

```js
// scripts/verify-against-pint.js
"use strict";
var fs = require("fs");
var path = require("path");
var convert = require("../lib");

var refPath = path.join(__dirname, "pint-reference.json");
if (!fs.existsSync(refPath)) {
  console.error("pint-reference.json not found. Run: python3 scripts/gen-pint-reference.py");
  process.exit(1);
}

var reference = JSON.parse(fs.readFileSync(refPath, "utf8"));
var results = { exact: 0, acceptable: 0, review: 0, wrong: 0, skipped: 0 };

Object.keys(reference).forEach(function (measure) {
  Object.keys(reference[measure]).forEach(function (abbr) {
    var ref = reference[measure][abbr];
    if (ref.error) {
      results.skipped++;
      return;
    }

    try {
      var actual = convert(1).from(abbr).to(ref.anchor);
      var expected = ref.to_anchor_pint;
      var relError = Math.abs((actual - expected) / expected);

      var status;
      if (relError === 0) {
        status = "EXACT";
        results.exact++;
      } else if (relError < 1e-6) {
        status = "ACCEPTABLE";
        results.acceptable++;
      } else if (relError < 1e-3) {
        status = "REVIEW";
        results.review++;
        console.log("  REVIEW: " + abbr + " (" + measure + ") — error: " + relError.toExponential(2) +
          " (convert-units: " + actual + ", pint: " + expected + ")");
      } else {
        status = "WRONG";
        results.wrong++;
        console.error("  WRONG: " + abbr + " (" + measure + ") — error: " + relError.toExponential(2) +
          " (convert-units: " + actual + ", pint: " + expected + ")");
      }
    } catch (e) {
      results.skipped++;
    }
  });
});

var total = results.exact + results.acceptable + results.review + results.wrong;
console.log("\nPint cross-validation: " + total + " units compared");
console.log("  EXACT:      " + results.exact);
console.log("  ACCEPTABLE: " + results.acceptable + " (< 1e-6 relative error)");
console.log("  REVIEW:     " + results.review + " (1e-6 to 1e-3)");
console.log("  WRONG:      " + results.wrong + " (> 1e-3)");
console.log("  SKIPPED:    " + results.skipped);

if (results.wrong > 0 || results.review > 0) {
  process.exit(1);
}
```
